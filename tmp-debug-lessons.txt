import { MongoClient } from 'mongodb';
import dotenv from 'dotenv';
dotenv.config({ path: '.env.local' });

async function run() {
    const uri = process.env.MONGODB_URI as string;
    const client = new MongoClient(uri);
    await client.connect();
    const db = client.db();

    const student = await db.collection('users').findOne({ role: 'student' });
    console.log("== STUDENT ==");
    if (student) {
        console.log(`Name: ${student.name}`);
        console.log(`Grade: '${student.grade}' (length: ${student.grade?.length})`);
        console.log(`Status: ${student.status}`);
    } else {
        console.log("NO STUDENT FOUND");
    }

    const lessons = await db.collection('lessons').find({}).toArray();
    console.log(`\n== LESSONS (Total: ${lessons.length}) ==`);
    lessons.forEach(l => {
        console.log(`- Title: ${l.title}`);
        console.log(`  Grade: '${l.grade}' (length: ${l.grade?.length}) - Matches student? ${student?.grade === l.grade}`);
        console.log(`  isPublished: ${l.isPublished}`);
        console.log(`  Subject: '${l.subject}'`);
    });

    process.exit(0);
}

run().catch(console.error);
