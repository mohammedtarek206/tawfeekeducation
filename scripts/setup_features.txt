import fs from 'fs';
import path from 'path';

const root = 'd:/مشاربع/التوفيق منصه/app';

// 1. Admin Pages
const adminExamsPage = fs.readFileSync(path.join(root, 'admin/exams/page.tsx'), 'utf-8');

function extractForm(title: string, type: string, description: string) {
    let replaced = adminExamsPage.replace(
        /الامتحانات والكويزات/g,
        title
    ).replace(
        /إنشاء وإدارة امتحانات أسبوعية وشهرية وكويزات الحصص/g,
        description
    ).replace(
        /type: 'weekly'/g,
        `type: '${type}'`
    );
    // Remove the select for type
    replaced = replaced.replace(
        /<label className="input-label">النوع<\/label>[\s\S]*?<\/select>/,
        `<label className="input-label">النوع</label><input className="input-field bg-gray-100" readOnly value="${type}" />`
    );
    return replaced;
}

const sections = [
    { dir: 'admin/lesson-quizzes', code: extractForm('اختبارات الحصص', 'quiz', 'إدارة اختبارات الحصص المرتبطة بالدروس') },
    { dir: 'admin/weekly-exams', code: extractForm('الاختبارات الأسبوعية', 'weekly', 'إدارة الاختبارات الأسبوعية للمراجعة الدورية') },
    { dir: 'admin/monthly-exams', code: extractForm('الاختبارات الشهرية', 'monthly', 'إدارة الاختبارات الشهرية الشاملة') },
];

sections.forEach(s => {
    const dirPath = path.join(root, s.dir);
    if (!fs.existsSync(dirPath)) fs.mkdirSync(dirPath, { recursive: true });
    fs.writeFileSync(path.join(dirPath, 'page.tsx'), s.code);
});
